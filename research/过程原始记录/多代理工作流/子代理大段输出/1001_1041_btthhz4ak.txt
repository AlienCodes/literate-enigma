    46	def _free_port(preferred: int) -> int:
    47	    for port in [preferred] + list(range(preferred + 1, preferred + 50)):
    48	        with socket.socket(socket.AF_INET, socket.SOCK_STREAM) as s:
    49	            try:
    50	                s.bind(("127.0.0.1", port))
    51	                return port
    52	            except OSError:
    53	                continue
    54	    return preferred

    94	    def check(self) -> List[str]:
    95	        problems = []
    96	        if self.external_url:
    97	            return problems
    98	        if not self.root or not self.root.exists():
    99	            return [f"找不到 GPT-SoVITS 目录：{self.root}（请运行安装脚本，或在 config.yaml 里设置 backends.gptsovits.root）"]
   100	        if not (self.root / "api_v2.py").exists():
   101	            problems.append(f"{self.root} 不是完整的 GPT-SoVITS 目录（缺少 api_v2.py）")

   103	        for rel in missing:
   104	            problems.append(f"缺少预训练模型：{rel}")
   105	        if missing:
   106	            problems.append("可运行 voicetwin download-models 自动下载（国内加 --source hf-mirror）")
   107	        return problems

   116	    def download_pretrained(self, source: str = "auto", progress: Optional[ProgressFn] = None) -> List[str]:
   117	        """从 Hugging Face（lj1995/GPT-SoVITS）下载缺失的预训练模型；国内可用 source="hf-mirror"。"""
   118	        import requests
   119	
   120	        if not self.root or not self.root.exists():
   121	            raise RuntimeError(f"找不到 GPT-SoVITS 目录：{self.root}")
   122	        endpoints = {"hf": ["https://huggingface.co"], "hf-mirror": ["https://hf-mirror.com"]}.get(
   123	            source, [os.environ.get("HF_ENDPOINT", "https://huggingface.co").rstrip("/"), "https://hf-mirror.com"])
   124	        repo = "lj1995/GPT-SoVITS"
   125	        prefix = "GPT_SoVITS/pretrained_models/"
   126	        missing = self.missing_pretrained()
   127	        if not missing:
   128	            log.info("GPT-SoVITS 预训练模型齐全，无需下载")
   129	            return []
   130	        session = requests.Session()
   131	        last_error: Optional[Exception] = None
   132	        for ep in endpoints:
   133	            try:
   134	                files: List[str] = []
   135	                for rel in missing:
   136	                    remote = rel[len(prefix):]
   137	                    if rel in (BERT_DIR, HUBERT_DIR):
   138	                        r = session.get(f"{ep}/api/models/{repo}/tree/main/{remote}", timeout=30)
   139	                        r.raise_for_status()
   140	                        files += [item["path"] for item in r.json() if item.get("type") == "file"]
   141	                    else:
   142	                        files.append(remote)
   143	                for i, remote in enumerate(files):
   144	                    dst = self.p(prefix + remote)
   145	                    if dst.exists():
   146	                        continue
   147	                    dst.parent.mkdir(parents=True, exist_ok=True)
   148	                    tmp = dst.with_suffix(dst.suffix + ".part")
   149	                    log.info(f"下载 {remote}（{ep}）……")
   150	                    with session.get(f"{ep}/{repo}/resolve/main/{remote}", stream=True, timeout=60) as r:
   151	                        r.raise_for_status()
   152	                        total = int(r.headers.get("Content-Length", 0))
   153	                        done = 0
   154	                        with open(tmp, "wb") as f:
   155	                            for chunk in r.iter_content(chunk_size=1 << 20):
   156	                                f.write(chunk)
   157	                                done += len(chunk)
   158	                                if progress and total:
   159	                                    progress((i + done / total) / len(files), f"{remote} {done >> 20}/{total >> 20} MB")
   160	                    tmp.replace(dst)
   161	                return files
   162	            except Exception as exc:
   163	                last_error = exc
   164	                log.warning(f"从 {ep} 下载失败：{exc}")
   165	        raise RuntimeError(f"预训练模型下载失败：{last_error}。也可以手动从 https://huggingface.co/{repo} 下载后放入 "
   166	                           f"{self.p(prefix)}")

   168	    def env(self, extra: Optional[Dict[str, Any]] = None) -> Dict[str, str]:
   169	        assert self.root is not None
   170	        root = str(self.root)
   171	        paths = [root, os.path.join(root, "GPT_SoVITS"), os.path.join(root, "GPT_SoVITS", "BigVGAN"),
   172	                 os.path.join(root, "tools"), os.path.join(root, "tools", "asr"), os.path.join(root, "tools", "uvr5")]
   173	        existing = os.environ.get("PYTHONPATH", "")
   174	        # 整合包的 ffmpeg.exe 在根目录，Python 在 runtime/，与官方 go-webui.bat 一样加入 PATH
   175	        bin_paths = [root, os.path.join(root, "runtime")]
   176	        env = subprocess_env({
   177	            "PATH": os.pathsep.join(bin_paths + [os.environ.get("PATH", "")]),
   178	            "PYTHONPATH": os.pathsep.join(paths + ([existing] if existing else [])),
   179	            "version": self.version,
   180	            "is_half": str(self.is_half),
   181	            "no_proxy": "localhost, 127.0.0.1, ::1",
   182	            "KMP_DUPLICATE_LIB_OK": "TRUE",
   183	        })
   184	        env.pop("all_proxy", None)
   185	        env.pop("ALL_PROXY", None)
   186	        if extra:
   187	            env.update({k: str(v) for k, v in extra.items()})
   188	        return env

   190	    def ensure_users_pth(self) -> None:
   191	        """和官方 webui.py 启动时一样，在 GPT-SoVITS 的 site-packages 写入 users.pth（整合包的训练脚本依赖它）。"""
   192	        assert self.root is not None
   193	        root = str(self.root).replace("\\", "/")
   194	        code = (
   195	            "import site, os\n"
   196	            f"root = {root!r}\n"
   197	            "content = '\\n'.join([root, root + '/GPT_SoVITS/BigVGAN', root + '/tools', root + '/tools/asr', "
   198	            "root + '/GPT_SoVITS', root + '/tools/uvr5'])\n"
   199	            "cands = [p for p in site.getsitepackages() if 'packages' in p] or [root + '/runtime/Lib/site-packages']\n"
   200	            "for sp in cands:\n"
   201	            "    if os.path.isdir(sp):\n"
   202	            "        f = os.path.join(sp, 'users.pth')\n"
   203	            "        try:\n"
   204	            "            if not os.path.exists(f) or open(f, encoding='utf-8', errors='ignore').read() != content:\n"
   205	            "                open(f, 'w', encoding='utf-8').write(content)\n"
   206	            "            break\n"
   207	            "        except OSError:\n"
   208	            "            pass\n"
   209	        )
   210	        try:
   211	            subprocess.run([self.python, "-c", code], env=self.env(), cwd=str(self.root), timeout=120,
   212	                           stdout=subprocess.DEVNULL, stderr=subprocess.DEVNULL)
   213	        except Exception as exc:  # pragma: no cover
   214	            log.warning(f"写入 users.pth 失败（通常不影响）：{exc}")

   220	    def _prepare_features(self, list_path: Path, wav_dir: Path, progress: Optional[ProgressFn]) -> Path:
   221	        opt_dir = self.p(f"logs/{self.exp_name}")
   222	        opt_dir.mkdir(parents=True, exist_ok=True)
   223	        digest = hashlib.sha1(list_path.read_bytes() + self.version.encode()).hexdigest()
   224	        stamp = opt_dir / "voicetwin_list.sha1"
   225	        if stamp.exists() and stamp.read_text().strip() != digest:
   226	            log.info("训练素材有变化，清理旧的特征文件后重新提取")
   227	            for name in ("2-name2text.txt", "6-name2semantic.tsv", "3-bert", "4-cnhubert", "5-wav32k", "7-sv_cn"):
   228	                target = opt_dir / name
   229	                if target.is_dir():
   230	                    shutil.rmtree(target, ignore_errors=True)
   231	                elif target.exists():
   232	                    target.unlink()
   233	        stamp.write_text(digest)
   234	        base = {"inp_text": str(list_path), "inp_wav_dir": str(wav_dir), "exp_name": self.exp_name,
   235	                "opt_dir": str(opt_dir), "i_part": "0", "all_parts": "1", "_CUDA_VISIBLE_DEVICES": self._gpu(),
   236	                "is_half": str(self.is_half), "version": self.version}
   237	
   238	        # 1A：文本 → 音素 + BERT 特征
   239	        path_text = opt_dir / "2-name2text.txt"
   240	        if not path_text.exists() or len(path_text.read_text(encoding="utf-8").strip().splitlines()) < 2:
   241	            self.step(progress, 0.05, "1A 文本处理（音素 + BERT 特征）")
   242	            self.run_logged([self.python, "-s", "GPT_SoVITS/prepare_datasets/1-get-text.py"], self.root,
   243	                            self.env({**base, "bert_pretrained_dir": str(self.p(BERT_DIR))}), "gsv_1a_text")
   244	            part = opt_dir / "2-name2text-0.txt"
   245	            lines = part.read_text(encoding="utf-8").strip("\n").split("\n") if part.exists() else []
   246	            if not "".join(lines).strip():
   247	                raise RuntimeError("1A 文本处理没有产出，请查看日志 logs/gsv_1a_text.log")
   248	            path_text.write_text("\n".join(lines) + "\n", encoding="utf-8")
   249	            part.unlink(missing_ok=True)
   250	
   251	        # 1B：HuBERT 特征 + 32k 音频（+ v2Pro 声纹）
   252	        self.step(progress, 0.12, "1B 提取 HuBERT 特征" + ("与说话人声纹" if "Pro" in self.version else ""))
   253	        env_1b = {**base, "cnhubert_base_dir": str(self.p(HUBERT_DIR)), "sv_path": str(self.p(SV_PATH))}
   254	        self.run_logged([self.python, "-s", "GPT_SoVITS/prepare_datasets/2-get-hubert-wav32k.py"], self.root,
   255	                        self.env(env_1b), "gsv_1b_hubert")
   256	        if "Pro" in self.version:
   257	            self.run_logged([self.python, "-s", "GPT_SoVITS/prepare_datasets/2-get-sv.py"], self.root,
   258	                            self.env(env_1b), "gsv_1b_sv")
   259	
   260	        # 1C：语义 token
   261	        path_sem = opt_dir / "6-name2semantic.tsv"
   262	        if not path_sem.exists() or path_sem.stat().st_size < 31:
   263	            self.step(progress, 0.2, "1C 提取语义 token")
   264	            env_1c = {**base, "pretrained_s2G": str(self.p(PRETRAINED_SOVITS[self.version])),
   265	                      "s2config_path": self._s2_config_template()}
   266	            self.run_logged([self.python, "-s", "GPT_SoVITS/prepare_datasets/3-get-semantic.py"], self.root,
   267	                            self.env(env_1c), "gsv_1c_semantic")
   268	            part = opt_dir / "6-name2semantic-0.tsv"
   269	            lines = ["item_name\tsemantic_audio"]
   270	            if part.exists():
   271	                lines += part.read_text(encoding="utf-8").strip("\n").split("\n")
   272	                part.unlink()
   273	            path_sem.write_text("\n".join(lines) + "\n", encoding="utf-8")
   274	        return opt_dir

   279	    def _auto_params(self, n_clips: int, minutes: float) -> Dict[str, int]:
   280	        tcfg = self.bcfg.get("train", {}) or {}
   281	        mem = gpu_memory_gb(self.python, self.env())
   282	        if mem <= 0:
   283	            log.warning("没有检测到可用的 NVIDIA 显卡，训练会非常慢（CPU 训练可能需要数天）")
   284	        bs = tcfg.get("batch_size", "auto")
   285	        if bs in (None, "auto"):
   286	            bs = int(mem // 2) if mem > 0 else 2
   287	        bs = max(1, min(int(bs), max(1, n_clips // 4)))
   288	        s_ep = tcfg.get("sovits_epochs", "auto")
   289	        if s_ep in (None, "auto"):
   290	            s_ep = 8 if minutes < 30 else 12
   291	        g_ep = tcfg.get("gpt_epochs", "auto")
   292	        if g_ep in (None, "auto"):
   293	            g_ep = 15
   294	        return {"batch_size": bs, "sovits_epochs": min(int(s_ep), 25), "gpt_epochs": int(g_ep), "gpu_mem_gb": round(mem, 1)}

   296	    def train(self, progress: Optional[ProgressFn] = None, **opts: Any) -> Dict[str, Any]:
   297	        from voicetwin.data.exporters import export_gptsovits
   298	
   299	        problems = self.check()
   300	        if self.external_url:
   301	            problems.append("配置了外部 api_url 时无法自动训练，请清空 backends.gptsovits.api_url 并设置 root")
   302	        if problems:
   303	            raise RuntimeError("GPT-SoVITS 环境有问题：\n- " + "\n- ".join(problems))
   304	        exp = export_gptsovits(self.project, speaker=self.exp_name)
   305	        log.info(f"训练素材：{exp['count']} 条，{exp['minutes']} 分钟；实验名 {self.exp_name}；版本 {self.version}")
   306	        params = self._auto_params(exp["count"], exp["minutes"])
   307	        params.update({k: v for k, v in opts.items() if v not in (None, "auto")})
   308	        tcfg = self.bcfg.get("train", {}) or {}
   309	        log.info(f"训练参数：batch={params['batch_size']}，SoVITS {params['sovits_epochs']} 轮，GPT {params['gpt_epochs']} 轮"
   310	                 f"（显存 {params['gpu_mem_gb']} GB）")
   311	        self.ensure_users_pth()
   312	        opt_dir = self._prepare_features(Path(exp["list"]), Path(exp["wav_dir"]), progress)
   313	
   314	        # ---------------- SoVITS
   315	        bs = int(params["batch_size"])
   316	        with open(self.p(self._s2_config_template()), "r", encoding="utf-8") as f:
   317	            s2 = json.load(f)
   318	        if not self.is_half:
   319	            s2["train"]["fp16_run"] = False
   320	            bs = max(1, bs // 2)
   321	        pre_g = str(self.p(PRETRAINED_SOVITS[self.version]))
   322	        s2["train"].update({
   323	            "batch_size": bs, "epochs": int(params["sovits_epochs"]),
   324	            "text_low_lr_rate": float(tcfg.get("text_low_lr_rate", 0.4)),
   325	            "pretrained_s2G": pre_g, "pretrained_s2D": pre_g.replace("s2G", "s2D"),
   326	            "if_save_latest": True, "if_save_every_weights": True,
   327	            "save_every_epoch": int(tcfg.get("sovits_save_every", 4)), "gpu_numbers": self._gpu(),
   328	            "grad_ckpt": False, "lora_rank": 32,
   329	        })
   330	        s2["model"]["version"] = self.version
   331	        s2["data"]["exp_dir"] = s2["s2_ckpt_dir"] = str(opt_dir)
   332	        s2["save_weight_dir"] = f"SoVITS_weights_{self.version}"
   333	        s2["name"] = self.exp_name
   334	        s2["version"] = self.version
   335	        (opt_dir / f"logs_s2_{self.version}").mkdir(parents=True, exist_ok=True)
   336	        s2_path = self.work_dir / "tmp_s2.json"
   337	        s2_path.write_text(json.dumps(s2, ensure_ascii=False), encoding="utf-8")
   338	        total_s = int(params["sovits_epochs"])
   339	        self.step(progress, 0.25, f"训练音色模型 SoVITS（共 {total_s} 轮）……")
   340	        self.run_logged([self.python, "-s", "GPT_SoVITS/s2_train.py", "--config", str(s2_path)], self.root, self.env(),
   341	                        "gsv_s2_train", progress, (0.25, 0.6), _epoch_parser(total_s))
   342	
   343	        # ---------------- GPT
   344	        with open(self.p("GPT_SoVITS/configs/s1longer-v2.yaml"), "r", encoding="utf-8") as f:
   345	            s1 = yaml.safe_load(f)
   346	        bs1 = int(params["batch_size"])
   347	        if not self.is_half:
   348	            s1["train"]["precision"] = "32"
   349	            bs1 = max(1, bs1 // 2)
   350	        s1["train"].update({
   351	            "batch_size": bs1, "epochs": int(params["gpt_epochs"]),
   352	            "save_every_n_epoch": int(tcfg.get("gpt_save_every", 5)), "if_save_every_weights": True,
   353	            "if_save_latest": True, "if_dpo": False, "half_weights_save_dir": f"GPT_weights_{self.version}",
   354	            "exp_name": self.exp_name,
   355	        })
   356	        s1["pretrained_s1"] = str(self.p(PRETRAINED_GPT[self.version]))
   357	        s1["train_semantic_path"] = str(opt_dir / "6-name2semantic.tsv")
   358	        s1["train_phoneme_path"] = str(opt_dir / "2-name2text.txt")
   359	        s1["output_dir"] = str(opt_dir / f"logs_s1_{self.version}")
   360	        (opt_dir / "logs_s1").mkdir(parents=True, exist_ok=True)
   361	        s1_path = self.work_dir / "tmp_s1.yaml"
   362	        s1_path.write_text(yaml.dump(s1, default_flow_style=False, allow_unicode=True), encoding="utf-8")
   363	        self.step(progress, 0.6, f"训练语气与节奏模型 GPT（共 {int(params['gpt_epochs'])} 轮）……")
   364	        self.run_logged([self.python, "-s", "GPT_SoVITS/s1_train.py", "--config_file", str(s1_path)], self.root,
   365	                        self.env({"_CUDA_VISIBLE_DEVICES": self._gpu(), "hz": "25hz"}), "gsv_s1_train",
   366	                        progress, (0.6, 0.95), _epoch_parser(int(params["gpt_epochs"])))
   367	
   368	        sovits, gpt = self._list_weights()
   369	        if not sovits or not gpt:
   370	            raise RuntimeError("训练结束但没有找到权重文件，请查看 logs/gsv_s2_train.log 与 logs/gsv_s1_train.log")
   371	        info = {
   372	            "version": self.version, "exp_name": self.exp_name, "trained_at": time.strftime("%Y-%m-%d %H:%M"),
   373	            "params": params, "sovits": [str(p) for p in sovits], "gpt": [str(p) for p in gpt],
   374	            "selected": {"id": f"s{_epoch(sovits[-1])}-g{_epoch(gpt[-1])}", "sovits": str(sovits[-1]), "gpt": str(gpt[-1])},
   375	        }
   376	        self.project.update_models(self.name, info)
   377	        self.step(progress, 1.0, "GPT-SoVITS 训练完成")
   378	        return info

   380	    def _list_weights(self) -> "tuple[List[Path], List[Path]]":
   381	        sov_dir = self.p(f"SoVITS_weights_{self.version}")
   382	        gpt_dir = self.p(f"GPT_weights_{self.version}")
   383	        sovits = sorted(sov_dir.glob(f"{self.exp_name}_e*_s*.pth"), key=_epoch) if sov_dir.exists() else []
   384	        gpt = sorted(gpt_dir.glob(f"{self.exp_name}-e*.ckpt"), key=_epoch) if gpt_dir.exists() else []
   385	        return sovits, gpt

   387	    def _locate_weight(self, path: str) -> Optional[Path]:
   388	        """models.json 里记的是绝对路径；整合包被移动 / 换了电脑后，按文件名到当前 root 里重新找。"""
   389	        if not path:
   390	            return None
   391	        p = Path(path)
   392	        if p.exists():
   393	            return p
   394	        if self.root is None:
   395	            return None
   396	        name = path.replace("\\", "/").rsplit("/", 1)[-1]
   397	        for d in (f"SoVITS_weights_{self.version}", f"GPT_weights_{self.version}"):
   398	            cand = self.p(d) / name
   399	            if cand.exists():
   400	                return cand
   401	        return None

   403	    def checkpoints(self) -> List[Dict[str, Any]]:
   404	        info = self.project.load_models().get(self.name) or {}
   405	        sovits = [w for w in (self._locate_weight(p) for p in info.get("sovits", [])) if w]
   406	        gpt = [w for w in (self._locate_weight(p) for p in info.get("gpt", [])) if w]
   407	        if not sovits or not gpt:
   408	            sovits, gpt = self._list_weights()
   409	        out = []
   410	        for s in sovits[-3:]:
   411	            for g in gpt[-3:]:
   412	                out.append({"id": f"s{_epoch(s)}-g{_epoch(g)}", "sovits": str(s), "gpt": str(g)})
   413	        return out

   417	        sel = self.selected_checkpoint()

   420	            if sovits and gpt:
   421	                return {"sovits": str(sovits), "gpt": str(gpt), "id": sel.get("id", "custom")}
   422	        if self.root is None:
   423	            return {"sovits": "", "gpt": "", "id": "external"}
   424	        log.warning("还没有训练好的 GPT-SoVITS 模型，暂时使用官方底模做零样本克隆（像度会明显低于训练后）")
   425	        return {"sovits": str(self.p(PRETRAINED_SOVITS[self.version])), "gpt": str(self.p(PRETRAINED_GPT[self.version])),
   426	                "id": "pretrained"}

   432	    def _session(self):
   433	        if self._http is None:
   434	            import requests
   435	
   436	            self._http = requests.Session()
   437	            self._http.trust_env = False  # 访问本机服务不走代理
   438	        return self._http

   440	    def _alive(self) -> bool:
   441	        """api_v2 对不带参数的 /tts 请求会返回 400 + {"message": ...}，以此判断服务已就绪。"""
   442	        try:
   443	            r = self._session().get(f"{self.api_url}/tts", timeout=3)
   444	            return r.status_code == 400 and "message" in r.json()
   445	        except Exception:
   446	            return False

   448	    def start(self) -> None:
   449	        if self._alive():
   450	            self._ensure_weights()
   451	            return
   452	        if self.external_url:
   453	            raise RuntimeError(f"连接不上 GPT-SoVITS 服务 {self.external_url}，请先启动 api_v2.py")
   454	        problems = self.check()
   455	        if problems:
   456	            raise RuntimeError("GPT-SoVITS 环境有问题：\n- " + "\n- ".join(problems))
   457	        self.port = _free_port(self.port)
   458	        self.api_url = f"http://127.0.0.1:{self.port}"
   459	        weights = self._current_weights()
   460	        cfg_path = self.work_dir / "tts_infer.yaml"
   461	        cfg_path.write_text(yaml.dump({"custom": {
   462	            "bert_base_path": str(self.p(BERT_DIR)), "cnhuhbert_base_path": str(self.p(HUBERT_DIR)),
   463	            "device": str(self.bcfg.get("device", "cuda")), "is_half": self.is_half, "version": self.version,
   464	            "t2s_weights_path": weights["gpt"], "vits_weights_path": weights["sovits"],
   465	        }}, allow_unicode=True), encoding="utf-8")
   466	        log_path = self.project.logs_dir / "gptsovits_api.log"
   467	        log.info(f"启动 GPT-SoVITS 推理服务（端口 {self.port}，模型 {weights['id']}）……")
   468	        creationflags = getattr(subprocess, "CREATE_NO_WINDOW", 0)
   469	        self._log_fh = open(log_path, "a", encoding="utf-8")
   470	        self.proc = subprocess.Popen(
   471	            [self.python, "api_v2.py", "-a", "127.0.0.1", "-p", str(self.port), "-c", str(cfg_path)],
   472	            cwd=str(self.root), env=self.env(), stdout=self._log_fh, stderr=subprocess.STDOUT, creationflags=creationflags,
   473	        )
   474	        deadline = time.time() + float(self.bcfg.get("startup_timeout", 600))
   475	        while time.time() < deadline:
   476	            if self.proc.poll() is not None:
   477	                self._log_fh.flush()
   478	                tail = log_path.read_text(encoding="utf-8", errors="replace").splitlines()[-30:]
   479	                raise RuntimeError("GPT-SoVITS 推理服务启动失败：\n" + "\n".join(tail))
   480	            if self._alive():
   481	                self._loaded = {"sovits": weights["sovits"], "gpt": weights["gpt"]}
   482	                log.info("GPT-SoVITS 推理服务已就绪")
   483	                return
   484	            time.sleep(2)
   485	        self.stop()
   486	        raise RuntimeError(f"GPT-SoVITS 推理服务启动超时，请查看 {log_path}")
   487	
   488	    def stop(self) -> None:
   489	        if self.proc is not None:
   490	            try:
   491	                self._session().get(f"{self.api_url}/control", params={"command": "exit"}, timeout=3)
   492	            except Exception:
   493	                pass
   494	            try:
   495	                self.proc.wait(timeout=10)
   496	            except Exception:
   497	                self.proc.kill()
   498	            self.proc = None
   499	            try:
   500	                self._log_fh.close()
   501	            except Exception:
   502	                pass
   503	        self._loaded = {}
   504	
   505	    def _set_weights(self, gpt: str, sovits: str) -> None:
   506	        s = self._session()
   507	        if gpt and self._loaded.get("gpt") != gpt:
   508	            r = s.get(f"{self.api_url}/set_gpt_weights", params={"weights_path": gpt}, timeout=600)
   509	            if r.status_code != 200:
   510	                raise RuntimeError(f"切换 GPT 模型失败：{r.text}")
   511	            self._loaded["gpt"] = gpt
   512	        if sovits and self._loaded.get("sovits") != sovits:
   513	            r = s.get(f"{self.api_url}/set_sovits_weights", params={"weights_path": sovits}, timeout=600)
   514	            if r.status_code != 200:
   515	                raise RuntimeError(f"切换 SoVITS 模型失败：{r.text}")
   516	            self._loaded["sovits"] = sovits
   517	
   518	    def _ensure_weights(self) -> None:
   519	        w = self._current_weights()
   520	        if w["sovits"] and w["gpt"]:
   521	            self._set_weights(w["gpt"], w["sovits"])

   523	    def use_checkpoint(self, ckpt: Dict[str, Any]) -> None:
   524	        if not self._alive():
   525	            self.start()
   526	        self._set_weights(ckpt["gpt"], ckpt["sovits"])

   528	    def synthesize(self, req: SynthRequest, out_path: Path) -> Path:
   529	        if not self._alive():
   530	            self.start()
   531	        icfg = self.bcfg.get("infer", {}) or {}
   532	        payload = {
   533	            "text": req.text,
   534	            "text_lang": "en" if req.lang == "en" else "zh",
   535	            "ref_audio_path": str(Path(req.ref_audio).resolve()),
   536	            "aux_ref_audio_paths": [str(Path(p).resolve()) for p in req.aux_refs],
   537	            "prompt_text": req.ref_text,
   538	            "prompt_lang": "en" if req.ref_lang == "en" else "zh",
   539	            "top_k": int(req.top_k if req.top_k is not None else icfg.get("top_k", 15)),
   540	            "top_p": float(req.top_p if req.top_p is not None else icfg.get("top_p", 1.0)),
   541	            "temperature": float(req.temperature if req.temperature is not None else icfg.get("temperature", 1.0)),
   542	            "text_split_method": "cut0",   # 已按句切好，不再二次切分，保留模型自己的句内停顿
   543	            "batch_size": 1,
   544	            "speed_factor": float(req.speed or 1.0),
   545	            "fragment_interval": 0.3,
   546	            "seed": int(req.seed),
   547	            "media_type": "wav",
   548	            "streaming_mode": False,
   549	            "parallel_infer": True,
   550	            "repetition_penalty": float(icfg.get("repetition_penalty", 1.35)),
   551	            "sample_steps": int(icfg.get("sample_steps", 32)),
   552	            "super_sampling": False,
   553	        }
   554	        r = self._session().post(f"{self.api_url}/tts", json=payload, timeout=600)
   555	        if r.status_code != 200:
   556	            try:
   557	                msg = r.json().get("message") or r.text
   558	            except Exception:
   559	                msg = r.text
   560	            raise RuntimeError(f"GPT-SoVITS 合成失败：{msg}")
   561	        out_path = Path(out_path)
   562	        out_path.parent.mkdir(parents=True, exist_ok=True)
   563	        out_path.write_bytes(r.content)
   564	        return out_path

   567	def _epoch(path: Path) -> int:
   568	    m = re.search(r"_e(\d+)_s\d+\.pth$", str(path)) or re.search(r"-e(\d+)\.ckpt$", str(path))
   569	    return int(m.group(1)) if m else 0

   571	
   572	def _epoch_parser(total: int):
   573	    pats = [re.compile(r"[Ee]poch[:=\s]*(\d+)"), re.compile(r"epoch=(\d+)")]
   574	
   575	    def parse(line: str) -> Optional[float]:
   576	        for pat in pats:
   577	            m = pat.search(line)
   578	            if m:
   579	                return min(1.0, int(m.group(1)) / max(total, 1))
   580	        return None
   581	
   582	    return parse
